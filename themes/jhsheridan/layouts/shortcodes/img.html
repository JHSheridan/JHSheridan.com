{{/* Styled image shortcode with optional caption Uses Hugo's resource system for automatic optimization Parameters: - src: image filename (required) - caption: image caption (optional) - alt: alt text (optional, falls back to caption or filename) - maxwidth: CSS max-width, default "635px" (optional) - float: "left" or "right" (optional) */}} {{- $src := .Get "src" -}} {{- $caption := .Get "caption" | default "" -}} {{- $alt := .Get "alt" | default $caption | default (path.BaseName $src) -}} {{- $maxwidth := .Get "maxwidth" | default "635px" -}} {{- $float := .Get "float" | default "" -}} {{/* Try to get image from page resources for optimization */}} {{- $image := .Page.Resources.GetMatch $src -}} {{- $imgSrc := $src -}} {{- if $image -}} {{/* Resize large images to max 1200px width, convert to WebP */}} {{- if gt $image.Width 1200 -}} {{- $image = $image.Resize "1200x webp" -}} {{- else -}} {{- $image = $image.Process "webp" -}} {{- end -}} {{- $imgSrc = $image.RelPermalink -}} {{- end -}}
